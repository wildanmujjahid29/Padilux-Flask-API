# ========================
# Port Scanning Detection
# ========================
alert tcp any any -> $HOME_NET any (
    msg:"[ALERT] Possible Port Scan Detected";
    flags:S;
    threshold:type threshold, track by_src, count 10, seconds 5;
    sid:100010;
    rev:1;
)

# ========================
# SSH Brute Force Attempt Detection
# ========================
alert tcp any any -> $HOME_NET 22 (
    msg:"[ALERT] SSH Brute Force Attempt";
    flags:S;
    threshold:type threshold, track by_src, count 5, seconds 10;
    sid:100011;
    rev:1;
)

# ========================
# SQL Injection Attempt Detection
# ========================
alert tcp any any -> $HOME_NET 80 (
    msg:"[CRITICAL] Possible SQL Injection Attempt";
    content:"select";
    nocase;
    content:"from";
    nocase;
    classtype:web-application-attack;
    sid:100012;
    rev:1;
)
